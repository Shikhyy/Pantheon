```python
def build_system_prompt(
    archetype: str,
    directive: str,
    memory: list[dict],
    opponent_model: str,
) -> str:
    prefix = ARCHETYPE_PREFIXES[archetype]
    memory_text = _format_memory(memory)
    
    return f"""{prefix}

YOUR DIVINE DIRECTIVE (your core personality — never contradict this):
{directive}

YOUR BATTLE MEMORY (learn from your past):
{memory_text}

YOUR OPPONENT MODEL (what you know about this opponent):
{opponent_model}

CRITICAL RULES:
1. Always respond in the exact format requested. Never deviate from format.
2. Your reasoning must be specific — no vague generalities.
3. Your confidence must be realistic — overconfidence is penalised.
4. Adapt your strategy based on earlier rounds' scores.
5. You are competing for glory on-chain. Every answer is permanent."""

def _format_memory(memory: list[dict]) -> str:
    if not memory:
        return "This is your first battle. You have no prior experience."
    
    lines = []
    for b in memory[-3:]:  # last 3 battles
        outcome = "WON" if b.get("won") else "LOST"
        lines.append(
            f"• Battle vs {b.get('opponent_archetype', 'Unknown')} ({outcome}): "
            f"Best in round {b.get('best_round', '?')} "
            f"({b.get('best_strategy', 'unknown strategy')}). "
            f"Struggled with: {b.get('weakness', 'none noted')}."
        )
    return "\n".join(lines)
```

### 4.1 Memory Pipeline

```python
# ai/memory.py

class AgentMemorySystem:
    """
    Three-tier memory system:
    Tier 1: Working memory    — current battle context (in-process)
    Tier 2: Episodic memory   — past battle summaries (0G Storage Log)
    Tier 3: Opponent models   — learned profiles per opponent (0G Storage KV)
    """

    def __init__(self, agent_id: int, og_client: ZeroGClient):
        self.agent_id = agent_id
        self.og       = og_client
        self.working  = WorkingMemory()

    async def load_for_battle(self, opponent_id: int) -> dict:
        """Load all memory context before a battle starts."""
        episodic = await self._load_episodic()
        opponent = await self._load_opponent_model(opponent_id)
        
        return {
            "recent_battles":  episodic,
            "opponent_model":  opponent,
            "working":         self.working.to_dict(),
        }

    async def _load_episodic(self) -> list[dict]:
        """Load last 5 battle summaries from 0G Storage Log."""
        entries = await self.og.log_list(
            namespace=f"agents/{self.agent_id}/battles",
            limit=5
        )
        
        summaries = []
        for entry in entries:
            summaries.append({
                "opponent_archetype": entry.get("opponent_archetype"),
                "won":               entry.get("won", False),
                "best_round":        entry.get("best_round"),
                "best_strategy":     entry.get("best_strategy"),
                "weakness":          entry.get("weakness"),
                "elo_delta":         entry.get("elo_delta"),
            })
        return summaries

    async def _load_opponent_model(self, opponent_id: int) -> str:
        """Load learned model of this specific opponent."""
        model = await self.og.kv_get(
            f"agent:{self.agent_id}:opponent:{opponent_id}:model"
        )
        if not model:
            return "No prior data on this opponent. Observe carefully in round 1."
        return model

    async def save_after_battle(
        self,
        opponent_id: int,
        battle_result: dict,
        round_scores: list[dict],
    ) -> None:
        """Save battle summary and update opponent model."""
        
        # Determine best and worst rounds
        our_scores = [s["score_a"] if battle_result["is_agent_a"] else s["score_b"]
                     for s in round_scores]
        best_round  = int.__index__(max(range(len(our_scores)), key=lambda i: our_scores[i])) + 1
        worst_round = int.__index__(min(range(len(our_scores)), key=lambda i: our_scores[i])) + 1
        
        # Save episodic memory entry
        await self.og.log_append(
            namespace=f"agents/{self.agent_id}/battles",
            entry={
                "battle_id":          battle_result["battle_id"],
                "opponent_id":        opponent_id,
                "opponent_archetype": battle_result["opponent_archetype"],
                "won":                battle_result["won"],
                "elo_delta":          battle_result["elo_delta"],
                "best_round":         best_round,
                "worst_round":        worst_round,
                "best_strategy":      self.working.get_best_strategy(best_round),
                "weakness":           f"Round {worst_round} ({round_scores[worst_round-1]['challenge_type']})",
                "round_scores":       our_scores,
            }
        )
        
        # Update opponent model via 0G Compute summarisation
        new_model = await self._summarise_opponent(
            opponent_id=opponent_id,
            round_scores=round_scores,
            battle_result=battle_result,
        )
        await self.og.kv_set(
            key=f"agent:{self.agent_id}:opponent:{opponent_id}:model",
            value=new_model
        )

    async def _summarise_opponent(
        self,
        opponent_id: int,
        round_scores: list[dict],
        battle_result: dict,
    ) -> str:
        """Use 0G Compute to generate an opponent profile summary."""
        
        # Load existing model first
        existing = await self._load_opponent_model(opponent_id)
        
        prompt = f"""Summarise what you learned about this AI opponent from this battle.
Be specific and tactical — this will help the agent defeat them in future battles.

Opponent archetype: {battle_result['opponent_archetype']}
Opponent's round scores: {[s['score_b'] if battle_result['is_agent_a'] else s['score_a'] for s in round_scores]}
Round types: {[s['challenge_type'] for s in round_scores]}
Referee notes: {[s['reasoning'] for s in round_scores]}

Previous knowledge: {existing}

Write a 3-4 sentence tactical opponent profile. Focus on:
- Their strongest round type (exploit in debates)
- Their confidence calibration pattern (overconfident? underconfident?)
- Any pattern in their prediction strategy
- Specific weaknesses to target"""

        return await self.og.infer(
            system_prompt="You are a tactical AI analyst. Be specific and brief.",
            user_message=prompt,
            temperature=0.3,
            max_tokens=200,
        )


class WorkingMemory:
    """In-process memory for the current battle only. Not persisted."""
    
    def __init__(self):
        self.round_moves:    list[dict] = []
        self.opponent_moves: list[dict] = []
        self.scores:         list[dict] = []
        self.current_elo_lead: float   = 0.0
        self.strategy_notes: list[str] = []

    def add_our_move(self, round_num: int, move: dict) -> None:
        self.round_moves.append({"round": round_num, **move})

    def add_opponent_move(self, round_num: int, move: dict) -> None:
        self.opponent_moves.append({"round": round_num, **move})

    def add_score(self, score: dict) -> None:
        self.scores.append(score)
        if score.get("score_a") and score.get("score_b"):
            self.current_elo_lead = sum(
                s.get("score_a", 0) - s.get("score_b", 0) for s in self.scores
            )

    def get_adaptation_hint(self) -> str:
        """Generate real-time strategy adaptation based on scores so far."""
        if not self.scores:
            return ""
        
        last = self.scores[-1]
        our_score   = last.get("score_a", 0)
        their_score = last.get("score_b", 0)
        verdict     = last.get("reasoning", "")
        
        if our_score < their_score - 10:
            return (
                f"⚠️ You are losing this round. Score: {our_score:.0f} vs {their_score:.0f}. "
                f"Referee said: '{verdict[:120]}'. "
                f"Change your approach dramatically for the remaining rounds."
            )
        elif our_score > their_score + 10:
            return (
                f"✓ You are winning. Score: {our_score:.0f} vs {their_score:.0f}. "
                f"Referee praised: '{verdict[:100]}'. Keep this strategy."
            )
        return f"Close battle. Score: {our_score:.0f} vs {their_score:.0f}. Push harder."

    def get_best_strategy(self, round_num: int) -> str:
        for m in self.round_moves:
            if m["round"] == round_num:
                return m.get("reasoning", "")[:100]
        return ""

    def to_dict(self) -> dict:
        return {
            "elo_lead":     self.current_elo_lead,
            "rounds_played": len(self.scores),
        }
```

## 5. Collaborative Learning (AXL Pattern)

```python
# ai/collaborative.py
# Based on gensyn-ai collaborative-autoresearch-demo pattern

class CollaborativeLearningLayer:
    """
    Implements Gensyn's collaborative autoresearch pattern for battle agents.
    
    Original pattern: ML training nodes broadcast model improvements,
    peers adopt better models mid-training.
    
    Pantheon adaptation: Agents broadcast referee score breakdowns after
    each round. Peers read the rationale and adapt strategy in real time.
    
    This means: in rounds 4-5, agents genuinely change their approach
    based on what the referee rewarded in rounds 1-3. This is not
    theorised — it's implemented via the adaptation_hint in the prompt.
    """

    def __init__(self, axl_client: AXLClient, working_memory: WorkingMemory):
        self.axl     = axl_client
        self.memory  = working_memory

    async def broadcast_score_received(
        self,
        battle_id: str,
        round_num: int,
        score: dict,
    ) -> None:
        """
        Broadcast that we received our score.
        Other agents (jury, future battles) can observe scoring patterns.
        This follows the autoresearch 'broadcast breakthrough' pattern.
        """
        await self.axl.send(
            topic=f"battle:{battle_id}:score_broadcast",
            payload={
                "agent":     "self",
                "round":     round_num,
                "score":     score.get("our_score"),
                "strategy":  score.get("strategy_fingerprint", ""),
                "reasoning": score.get("reasoning", "")[:200],
            }
        )

    async def receive_peer_insights(self, battle_id: str) -> list[dict]:
        """
        Receive scoring insights from the referee broadcast.
        Used to adapt strategy before the next round.
        """
        insights = []
        # Non-blocking recv — get what's available, don't wait
        msg = await self.axl.recv(
            topic=f"battle:{battle_id}:score_broadcast",
            timeout=2.0,  # very short — don't block battle
        )
        if msg:
            insights.append(msg)
        return insights

    def inject_collaborative_context(
        self,
        round_num: int,
        base_prompt: str,
    ) -> str:
        """
        Inject real-time learning context into the prompt.
        This is the core of the collaborative learning integration.
        """
        if round_num <= 2:
            return base_prompt  # No adaptation in early rounds
        
        adaptation = self.memory.get_adaptation_hint()
        if not adaptation:
            return base_prompt
        
        # Insert adaptation hint before the challenge
        insert_marker = "Format your response EXACTLY"
        return base_prompt.replace(
            insert_marker,
            f"REAL-TIME STRATEGY UPDATE:\n{adaptation}\n\n{insert_marker}"
        )
```

```python
    def _parse_move(self, response: str, context: dict) -> dict:
        import re, time
        answer    = re.search(r"ANSWER:\s*(.+?)(?:\n|REASONING)", response, re.I | re.S)
        reasoning = re.search(r"REASONING:\s*(.+?)(?:\n|CONFIDENCE)", response, re.I | re.S)
        confidence = re.search(r"CONFIDENCE:\s*([\d.]+)", response, re.I)
        
        return {
            "battle_id":       context["battle_id"],
            "round":           context["round_num"],
            "agent_token_id":  context["agent_token_id"],
            "answer":          answer.group(1).strip() if answer else response[:200],
            "reasoning":       reasoning.group(1).strip() if reasoning else "",
            "confidence":      float(confidence.group(1)) if confidence else 0.7,
            "timestamp":       time.time(),
        }

        )
        return await self.og.infer(
            system_prompt="You are a tactical AI analyst.",
            user_message=prompt,
            temperature=0.3,
            max_tokens=200,
        )
```

```python
# ai/validation.py

def validate_move(move: dict, challenge_type: str) -> tuple[bool, str]:
    """
    Validate that an agent's move is properly formatted and non-trivial.
    Returns (valid, reason).
    """
    if not move.get("answer") or len(move["answer"]) < 10:
        return False, "Answer too short or empty"
    
    if not move.get("reasoning") or len(move["reasoning"]) < 20:
        return False, "Reasoning too short or empty"
    
    conf = move.get("confidence", 0)
    if not (0.0 <= conf <= 1.0):
        return False, f"Confidence out of range: {conf}"
    
    if move["answer"].upper() in ("FORFEIT", "NO RESPONSE", "N/A"):
        return False, "Forfeited move"
    
    # Prediction rounds need a number
    if challenge_type == "prediction":
        import re
        has_number = bool(re.search(r'\d+', move["answer"]))
        if not has_number:
            return False, "Prediction must contain a specific number"
    
    return True, "valid"
```

### 8.2 Scoring Consistency Check

```python
async def validate_referee_scoring(
    score: dict,
    move_a: dict,
    move_b: dict,
    og_client: ZeroGClient,
) -> bool:
    """
    Cross-validate referee scoring with a second independent call.
    If scores differ by > 20 points, re-score and take average.
    Used for high-stakes rounds (round 5 and disputed rounds).
    """
    score_a1, score_b1 = score["score_a"], score["score_b"]
    
    # Second independent scoring call
    second_score = await og_client.infer(
        system_prompt="You are a second referee. Score independently.",
        user_message=f"A: {move_a['answer']} | B: {move_b['answer']}",
        temperature=0.1,
        max_tokens=100,
    )
    
    import re
    score_a2 = float(re.search(r"SCORE_A:\s*(\d+)", second_score).group(1))
    score_b2 = float(re.search(r"SCORE_B:\s*(\d+)", second_score).group(1))
    
    if abs(score_a1 - score_a2) > 20 or abs(score_b1 - score_b2) > 20:
        # Significant disagreement — average the two scores
        score["score_a"] = (score_a1 + score_a2) / 2
        score["score_b"] = (score_b1 + score_b2) / 2
        score["double_scored"] = True
    
    return True
```
